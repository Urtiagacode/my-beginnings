carnet = {
    "Alice": [18, 10],
    "Yanis": [9, 15],
    "Alex": [2, 12],
    "andoni": [14, 16]
}

for nom, notes in carnet.items():
    print(nom, notes)

def moyenne(notes):
    return sum(notes) / len(notes)

print("Moyenne de Alice :", moyenne(carnet["Alice"]))
print("Moyenne de Yanis :", moyenne(carnet["Yanis"]))
print("Moyenne de Alex :", moyenne(carnet["Alex"]))
print("Moyenne de andoni :", moyenne(carnet["andoni"]))


def meilleure_moyenne(carnet):
    meilleur_nom = None
    meilleure_valeur = 0

    for nom, notes in carnet.items():
        m = moyenne(notes)
        if meilleur_nom is None or m > meilleure_valeur:
            meilleur_nom = nom
            meilleure_valeur = m

    return meilleur_nom, meilleure_valeur

nom_meilleur, valeur_meilleure = meilleure_moyenne(carnet)
print("Meilleure moyenne :", nom_meilleur, valeur_meilleure)
